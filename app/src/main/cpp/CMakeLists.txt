cmake_minimum_required(VERSION 3.22.1)
project(brotherfastprint LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BRLASER ${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/brlaser/src)
add_library(fastprint SHARED bridge.cpp encoder.cpp mono.cpp ${BRLASER}/job.cc ${BRLASER}/line.cc)
target_include_directories(fastprint PRIVATE ${BRLASER})
target_compile_options(fastprint PRIVATE -Wall -Wextra -Werror=format-security -fstack-protector-strong)
target_link_options(fastprint PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,relro,-z,now)
target_link_libraries(fastprint PRIVATE jnigraphics)
